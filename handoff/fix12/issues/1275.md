call pgpm.transmute('public.f1jp', 'ts', interval '1 hour', p_obtain => 12);
\set ON_ERROR_STOP 1

          'transmute converts the Tokyo-wall-time table on an hourly grid with obtain 12 > the +9 offset, as reference.md describes');
